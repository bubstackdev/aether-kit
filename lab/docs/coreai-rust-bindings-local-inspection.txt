def _is_wheel_install() -> bool:
    # Is "coreai-core" installed?
    try:
        dist = _meta.distribution("coreai-core")
    except _meta.PackageNotFoundError:
        return False  # Not installed; running from the build tree via PYTHONPATH

    direct_url = dist.read_text("direct_url.json")
    if not direct_url:
        return True  # pip/uv omits direct_url.json

    try:
        info = _json.loads(direct_url)
    except _json.JSONDecodeError:
        # Something went wrong. Warn and assume wheel install.
        _log.warning("Assuming wheel install")
        return True

    # Based on PEP 610 `info` will have "archive_info" key if installed from wheel.
    return "archive_info" in info
